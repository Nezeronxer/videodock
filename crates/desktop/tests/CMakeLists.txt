cmake_minimum_required(VERSION 3.21)
project(videodock_appearance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2 Test WebEngineQuick Widgets Network)
qt_add_executable(videodock-appearance appearance.cpp)
set_target_properties(videodock-appearance PROPERTIES MACOSX_BUNDLE FALSE)
target_link_libraries(videodock-appearance PRIVATE
    Qt6::Quick Qt6::QuickControls2 Qt6::Test Qt6::WebEngineQuick Qt6::Widgets)
qt_add_executable(videodock-browser-proxy browser_proxy.cpp ../src/platform.cpp)
set_target_properties(videodock-browser-proxy PROPERTIES MACOSX_BUNDLE FALSE)
target_include_directories(videodock-browser-proxy PRIVATE ../include)
target_link_libraries(videodock-browser-proxy PRIVATE Qt6::Quick Qt6::WebEngineQuick Qt6::Widgets Qt6::Network)
if(MSVC)
    target_compile_options(videodock-browser-proxy PRIVATE /UNDEBUG)
else()
    target_compile_options(videodock-browser-proxy PRIVATE -UNDEBUG)
endif()
